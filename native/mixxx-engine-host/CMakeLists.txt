cmake_minimum_required(VERSION 3.21)
project(plumdeck_mixxx_host_seam LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Network)
add_executable(plumdeck-mixxx-host-seam src/main.cpp src/host.cpp src/unavailable_backend.cpp)
target_link_libraries(plumdeck-mixxx-host-seam PRIVATE Qt6::Core Qt6::Network plumdeck-junction-core)
if(MSVC)
  target_compile_options(plumdeck-mixxx-host-seam PRIVATE /W4 /utf-8)
else()
  target_compile_options(plumdeck-mixxx-host-seam PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_program(NODE_EXECUTABLE node)
if(NODE_EXECUTABLE)
  add_test(NAME protocol-seam COMMAND "${NODE_EXECUTABLE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol.test.mjs")
  set_tests_properties(protocol-seam PROPERTIES ENVIRONMENT "PLUMDECK_TEST_HOST=$<TARGET_FILE:plumdeck-mixxx-host-seam>")
endif()

include(cmake/junction.cmake)
add_executable(junction-core-tests tests/junction/harness.cpp tests/junction/clock_test.cpp tests/junction/media_transport_test.cpp tests/junction/asset_cache_test.cpp tests/junction/program_output_test.cpp tests/junction/producer_tap_test.cpp tests/junction/validation_capture_test.cpp tests/junction/authority_test.cpp tests/junction/network_settings_test.cpp tests/junction/manual_exchange_test.cpp tests/junction/shared_tracks_test.cpp)
target_link_libraries(junction-core-tests PRIVATE plumdeck-junction-core)
add_test(NAME junction-core COMMAND junction-core-tests)
